        // Charger un flux Blogger via JSONP (Bypasse le protocole file:// et toutes les erreurs CORS)
        function fetchBloggerFeedJSONP(blog) {
            return new Promise((resolve) => {
                const callbackName = 'cb_' + Math.random().toString(36).substring(2, 9);
                const script = document.createElement('script');
                
                const timer = setTimeout(() => {
                    delete window[callbackName];
                    if (script.parentNode) script.parentNode.removeChild(script);
                    resolve([]);
                }, 4000);

                window[callbackName] = function(data) {
                    clearTimeout(timer);
                    delete window[callbackName];
                    if (script.parentNode) script.parentNode.removeChild(script);

                    const entries = (data && data.feed && data.feed.entry) ? data.feed.entry : [];
                    const articles = entries.map((entry, entryIndex) => {
                        const publishedDate = new Date(entry.published.$t);
                        
                        let trimester = "T4";
                        const titleLower = entry.title.$t.toLowerCase();
                        if (blog.url.includes("t3") || titleLower.includes("marc") || entryIndex % 4 === 2) trimester = "T3";
                        else if (entryIndex % 4 === 1) trimester = "T2";
                        else if (entryIndex % 4 === 0) trimester = "T1";

                        const contentHtml = entry.content ? entry.content.$t : (entry.summary ? entry.summary.$t : "");
                        const tempDiv = document.createElement("div");
                        tempDiv.innerHTML = contentHtml;
                        const plainText = tempDiv.textContent || tempDiv.innerText || "";
                        const linkObj = entry.link ? entry.link.find(l => l.rel === 'alternate') : null;

                        return {
                            title: entry.title.$t,
                            blogName: blog.name,
                            blogUrl: blog.url,
                            pole: blog.pole,
                            date: publishedDate.toLocaleDateString('fr-FR', { day: 'numeric', month: 'long', year: 'numeric' }),
                            rawDate: publishedDate,
                            quarter: trimester,
                            summary: plainText.substring(0, 160) + "...",
                            content: contentHtml,
                            originalUrl: linkObj ? linkObj.href : blog.url
                        };
                    });
                    resolve(articles);
                };

                script.src = `${blog.url}/feeds/posts/default?alt=json-in-script&callback=${callbackName}&max-results=5`;
                script.onerror = () => {
                    clearTimeout(timer);
                    delete window[callbackName];
                    if (script.parentNode) script.parentNode.removeChild(script);
                    resolve([]);
                };

                document.head.appendChild(script);
            });
        }
